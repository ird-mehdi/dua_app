### SaveBookmarksToAyahUseCase
```
class SaveBookmarksToAyahUseCase extends BaseUseCase<String> {
  SaveBookmarksToAyahUseCase(
    super.errorMessageHandler,
    this._userDataRepository,
  );

  final UserDataRepository _userDataRepository;

  Future<Either<String, String>> execute({
    required int surahID,
    required int ayahID,
    required List<BookmarkFolderEntity> savingFolders,
  }) async {
    return mapResultToEither(() async {
      final List<BookmarkEntity> generatedBookmarks =
          await _mapBookmarkFoldersIntoBookmarks(savingFolders, surahID, ayahID);
      await _userDataRepository.saveBookmarksToAyah(
        surahID: surahID,
        ayahID: ayahID,
        bookmarks: generatedBookmarks,
      );
      return savingFolders.isNotEmpty ? "Collection saved" : "";
    });
  }
}
```

```
Future<void> saveBookmarksForAyah({
  required int surahID,
  required int ayahID,
  required void Function(int, {required bool isBookmarked}) onSaved,
  required BuildContext context,
}) async {
  final Set<String> selectedBookmarkFolderNames = currentUiState.selectedBookmarkFolderNames;
  
  uiState.value = uiState.value.copyWith(isBookmarkChanged: true);

  final String bookmarkSavedMessage = context.l10n.bookmarkSaved;
  final String bookmarkRemovedMessage = context.l10n.removeBookmark;

  final List<BookmarkFolderEntity> bookmarkFolders =
      currentUiState.bookmarkFolders.filter((f) => selectedBookmarkFolderNames.contains(f.name)).toList();

  await executeMessageOnlyUseCase(
    () async => _saveBookmarksToAyahUseCase.execute(
      surahID: surahID,
      ayahID: ayahID,
      savingFolders: bookmarkFolders,
    ),
    onSuccess: () async {
      await syncCollectionsWithRemote();

      onSaved(
        bookmarkFolders.length,
        isBookmarked: bookmarkFolders.isNotEmpty,
      );
      
      if (bookmarkFolders.isEmpty) {
        addUserMessage(bookmarkRemovedMessage);
      } else if (!_wasBookmarkJustAdded || selectedBookmarkFolderNames.length > 1) {
        addUserMessage(bookmarkSavedMessage);
      }
    },
    showMessage: false,
  );

  _wasBookmarkJustAdded = false;
  await _clearSelection();
}
```

```
Future<bool> updateBookmarkFolder({
  required BookmarkFolderEntity folder,
  required String newName,
  required Color newColor,
}) async {
  final bool isValid = folder.name == newName ||
      await _validateFolderName(
        name: newName.trim(),
      );
  if (!isValid) return false;
  await executeMessageOnlyUseCase(
    () async => _updateBookmarkFolderUseCase.execute(
      folderName: folder.name.trim(),
      newFolderName: newName.trim(),
      color: newColor,
    ),
    onSuccess: () async {
      await syncCollectionsWithRemote();
    },
  );
  await fetchBookmarks();
  return true;
}
```

```
Future<void> deleteBookmarkFolder({
  required BookmarkFolderEntity folder,
}) async {
  await parseDataFromEitherWithUserMessage(
    task: () => _deleteBookmarkFolderUseCase.execute(folder: folder),
    onDataLoaded: (message) async {
      final List<BookmarkFolderEntity> bookmarks = currentUiState.bookmarkFolders.toList(growable: true)
        ..remove(folder);

      if (_searchQuery.isNotEmpty) {
        clearSearch();
      }

      uiState.value = currentUiState.copyWith(bookmarkFolders: bookmarks);
      await syncCollectionsWithRemote();
      await fetchBookmarks();
      await addUserMessage(message);
    },
  );
}
```

```
Future<void> searchBookmarks({required String query}) async {
  final String trimmedQuery = query.trim();
  _searchQuery = trimmedQuery;

  if (trimmedQuery.isEmpty) {
    await fetchBookmarks();
    return;
  }

  if (trimmedQuery == _lastBookmarkQuery) return;

  await parseDataFromEitherWithUserMessage(
    task: () => _searchBookmarkUseCase.execute(
      query: trimmedQuery,
      allFolders: _allBookmarkFolders,
    ),
    onDataLoaded: (bookmarks) {
      uiState.value = currentUiState.copyWith(
        filteredBookmarkFolders: bookmarks,
      );
    },
  );
  _lastBookmarkQuery = trimmedQuery;
}
```

```
Future<void> addAyahToBookmarkFolder({
  required BookmarkEntity bookmark,
  bool isImport = false,
}) async {
  return synchronized(() async {
    if (!isImport) {
      final existingBookmark = await _userDataStorage.getBookmarksOfSameFolderWithSameAyah(
        surahID: bookmark.surahID,
        folderName: bookmark.folderName,
        ayahID: bookmark.ayahID,
      );

      if (existingBookmark != null) {
        return; // Skip if already exists
      }
    }

    final String colorHex = getHexFromColor(bookmark.color);

    await _userDataStorage.addAyahToBookmarkFolder(
      surahID: bookmark.surahID,
      ayahID: bookmark.ayahID,
      folderName: bookmark.folderName,
      color: colorHex,
      createdAt: bookmark.createdAt,
      updatedAt: bookmark.updatedAt,
    );
  });
}
```

```
Future<void> saveBookmarksToAyah({
  required int surahID,
  required int ayahID,
  required List<BookmarkEntity> bookmarks,
}) async {
  await catchFutureOrVoid(() async {
    await deleteAyahFromBookmark(surahID: surahID, ayahID: ayahID);
    if (bookmarks.isEmpty) return;

    final List<Map<String, Object?>> bookmarkMaps = await Future.wait(bookmarks.map((b) async => b
        .copyWith(
          createdAt: b.createdAt,
          updatedAt: b.updatedAt,
        )
        .toMap()));

    await _backendAsAService.saveBookmarks(bookmarks: bookmarkMaps);
  });
}
```
